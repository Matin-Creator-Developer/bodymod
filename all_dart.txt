  int get targetDailyCalories {
    // فرمول Mifflin-St Jeor
    final int age = (1405 - (int.tryParse(birthYear) ?? 1383)).clamp(12, 90);
    double bmr = (10 * weightKg) + (6.25 * heightCm) - (5 * age);
    bmr += (gender == 'مرد') ? 5 : -161;
    double tdee = bmr * 1.45;
    if (fitnessGoal.contains('کاهش وزن')) return (tdee - 450).round();
    if (fitnessGoal.contains('افزایش وزن') || fitnessGoal.contains('عضله')) {
      return (tdee + 400).round();
    }
    return tdee.round();
  }

    final hMeter = heightCm / 100.0;

  // سهمیه‌بندی دقیق روزانه هوش مصنوعی:
  // در چت: روزانه ۳ پیام با Gemini 3.8 Flash و ۶ پیام با Gemini 3.5 Flash-Lite
  // در سایر بخش‌ها (برنامه غذایی، باشگاه، لاگ بدن، اسکنر): روزانه ۱ بار با 3.8 Flash و ۲ بار با 3.5 Flash-Lite
  int getQuotaLimit(String featureKey, String modelId) {
    final bool is38 = modelId.contains('3.8');
    if (featureKey == 'chat') {
      return is38 ? 3 : 6;
    } else {
      return is38 ? 1 : 2;

  }

  int getUsedQuotaToday(String featureKey, String modelId) {
    final bool is38 = modelId.contains('3.8');
    final String mTag = is38 ? '38' : '35lite';
    final String key = 'bm_quota_${todayKey()}_${featureKey}_$mTag';
    return prefs?.getInt(key) ?? 0;
  }

  int getRemainingQuotaToday(String featureKey, String modelId) {
    final limit = getQuotaLimit(featureKey, modelId);
    final used = getUsedQuotaToday(featureKey, modelId);
    return (limit - used).clamp(0, limit);
  }

  Future<void> incrementQuotaUsage(String featureKey, String modelId) async {
    final bool is38 = modelId.contains('3.8');
    final String mTag = is38 ? '38' : '35lite';
    final String key = 'bm_quota_${todayKey()}_${featureKey}_$mTag';
    final used = getUsedQuotaToday(featureKey, modelId);
    await prefs?.setInt(key, used + 1);
    notifyListeners();
  }

  // فراخوانی یکپارچه هوش مصنوعی از سرور کلودفلر با تزریق خودکار کل فرم پروفایل

    required String featureKey, // 'chat' | 'meal' | 'workout' | 'body_log' | 'scanner'
    required String modelId, // 'gemini-3.5-flash-lite' | 'gemini-3.8-flash'
    required String userPrompt,
  }) async {
    final remaining = getRemainingQuotaToday(featureKey, modelId);
    final modelDisplayName =
        modelId.contains('3.8') ? 'Gemini 3.8 Flash' : 'Gemini 3.5 Flash-Lite';

    if (remaining <= 0) {
      throw Exception(
        'سهمیه امروزت برای مدل $modelDisplayName در این بخش تموم شده رفیق! ⏳\n'
        'می‌تونی از منوی بالای همین بخش، مدل دیگه رو انتخاب کنی یا فردا دوباره امتحان کنی.',
      );
    }

    final fullPrompt = '${buildSystemProfileContext()}\n\nدرخواست کاربر:\n$userPrompt';

    final response = await http
        .post(
          Uri.parse('$kServerUrl/api/ai'),
          headers: {
            'Content-Type': 'application/json',
            if (token.isNotEmpty) 'Authorization': 'Bearer $token',
          },
          body: jsonEncode({
            'feature': featureKey,
            'model': modelId,
            'prompt': fullPrompt,
            'profile': buildFullProfileJson(),
          }),
        )
        .timeout(const Duration(seconds: 50));

    final data = jsonDecode(utf8.decode(response.bodyBytes));
    if (response.statusCode != 200 || data['error'] != null) {
      throw Exception(data['error'] ?? 'خطا در ارتباط با هوش مصنوعی بادی‌مود');
    }

    await incrementQuotaUsage(featureKey, modelId);
    return (data['reply'] ?? data['result'] ?? data['text'] ?? '').toString();
  }

  Timer? _countdownTimer;

  @override
  void dispose() {
    _countdownTimer?.cancel();
    _phoneCtrl.dispose();
    _otpCtrl.dispose();
    super.dispose();
  }

  void _start30SecTimer() {
    _countdownTimer?.cancel();
    setState(() => _timerSeconds = 30);
    _countdownTimer = Timer.periodic(const Duration(seconds: 1), (t) {
      if (_timerSeconds <= 1) {
        t.cancel();
        if (mounted) setState(() => _timerSeconds = 0);
      } else {
        if (mounted) setState(() => _timerSeconds--);
      }
    });
  }

  Future<void> _handleNativeGoogleSignIn() async {
    setState(() => _loading = true);
    try {
      final googleSignIn = GoogleSignIn(
        scopes: ['email', 'profile'],
        serverClientId: kGoogleWebClientId,
      );
      final account = await googleSignIn.signIn();
      if (account == null) {
        setState(() => _loading = false);
        return;
      }
      final auth = await account.authentication;
      final res = await http.post(
        Uri.parse('$kServerUrl/api/auth/google-native'),
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'email': account.email,
          'displayName': account.displayName ?? '',
          'idToken': auth.idToken ?? '',
          'accessToken': auth.accessToken ?? '',
        }),
      );

      if (res.statusCode == 200) {
        final data = jsonDecode(utf8.decode(res.bodyBytes));
        await widget.appState.setAuthSession(
          newToken: data['token'] ?? 'google_${account.id}',
          identifier: account.email,
          admin: data['isAdmin'] == true,
          pro: data['isPro'] == true,
        );
      } else {
        // در صورت عدم پاسخ سرور، ورود بومی با ایمیل گوگل تاییدشده انجام می‌شود
        await widget.appState.setAuthSession(
          newToken: 'google_native_${account.id}',
          identifier: account.email,
        );
      }
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('خطا در ورود بومی گوگل: $e')),
        );
      }
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({'phone': phone}),
      );

  final TextEditingController _msgCtrl = TextEditingController();
  final ScrollController _scrollCtrl = ScrollController();
  bool _sending = false;

  @override
  void dispose() {
    _msgCtrl.dispose();
    _scrollCtrl.dispose();
    super.dispose();
  }

  Future<void> _sendMessage() async {
    final st = widget.appState;
    final text = _msgCtrl.text.trim();
    if (text.isEmpty || _sending) return;

    _msgCtrl.clear();
    st.chatMessages.add({
      'role': 'user',
      'text': text,
      'model': st.chatModel,
    });
    setState(() => _sending = true);

    try {
      // ارسال سوال کاربر به همراه کل پرونده بدنی (که داخل callBodyModAI خودکار تزریق می‌شود)
      final reply = await st.callBodyModAI(
        featureKey: 'chat',
        modelId: st.chatModel,
        userPrompt: text,
      );
      st.chatMessages.add({
        'role': 'ai',
        'text': reply,
        'model': st.chatModel,
      });
      if (st.chatMessages.length > 40) {
        st.chatMessages = st.chatMessages.sublist(st.chatMessages.length - 40);
      }
      await st.prefs?.setString('bm_chat_history', jsonEncode(st.chatMessages));
    } catch (e) {
      st.chatMessages.add({
        'role': 'ai',
        'text': '⚠️ $e',
        'model': st.chatModel,
      });

      if (mounted) {
        setState(() => _sending = false);

          if (_scrollCtrl.hasClients) {
            _scrollCtrl.animateTo(
              _scrollCtrl.position.maxScrollExtent,

              curve: Curves.easeOut,

class _ProfileTabState extends State<ProfileTab> {

  late TextEditingController _weightCtrl;
  late TextEditingController _memoryCtrl;

  late String _bMonth;
  late String _bYear;
  late String _gender;
  late String _econ;
  late String _goal;
  late String _timeline;
  late String _workoutDays;

  final List<String> _daysList =
      List.generate(31, (i) => (i + 1).toString().padLeft(2, '0'));
  final List<String> _monthsList =

  static const List<String> _goalOptions = [
    'کاهش وزن و چربی‌سوزی',
    'افزایش وزن و حجم',
    'عضله‌سازی خشک و فیتنس',
    'تناسب اندام و سلامتی',
  ];
  static const List<String> _timelineOptions = [
    '۱ ماهه (فشرده)',
    '۳ ماهه',
    '۶ ماهه',
    '۱ ساله (اصولی و ماندگار)',
  ];
  static const List<String> _workoutDaysOptions = [
    '۱ روز در هفته',
    '۲ روز در هفته',
    '۳ روز در هفته',
    'هر روز',
  ];

  @override
  void initState() {
    super.initState();
    final st = widget.appState;
    _nickCtrl = TextEditingController(text: st.nickname);
    _cityCtrl = TextEditingController(text: st.city);
    _heightCtrl = TextEditingController(text: st.heightCm.toStringAsFixed(0));
    _weightCtrl = TextEditingController(text: st.weightKg.toStringAsFixed(1));
    _memoryCtrl = TextEditingController(text: st.aiMemoryNotes);

    _bDay = _daysList.contains(st.birthDay) ? st.birthDay : '14';
    _bMonth = _monthsList.contains(st.birthMonth) ? st.birthMonth : '10';
    _bYear = _yearsList.contains(st.birthYear) ? st.birthYear : '1383';
    _gender = (st.gender == 'زن') ? 'زن' : 'مرد';
    _econ = _econOptions.contains(st.economicStatus) ? st.economicStatus : 'متوسط';
    _goal = _goalOptions.contains(st.fitnessGoal)
        ? st.fitnessGoal
        : 'تناسب اندام و سلامتی';
    _timeline = _timelineOptions.contains(st.targetTimeline)
        ? st.targetTimeline
        : '۳ ماهه';
    _workoutDays = _workoutDaysOptions.contains(st.workoutDaysPerWeek)
        ? st.workoutDaysPerWeek
        : '۳ روز در هفته';
  }

  @override
  void dispose() {
    _nickCtrl.dispose();
    _cityCtrl.dispose();
    _heightCtrl.dispose();
    _weightCtrl.dispose();
    _memoryCtrl.dispose();
    super.dispose();
  }

  Future<void> _saveForm() async {
    final h = double.tryParse(_heightCtrl.text.trim()) ?? 175;
    final w = double.tryParse(_weightCtrl.text.trim()) ?? 70;
    if (h < 100 || h > 240 || w < 25 || w > 250) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('قد و وزن رو در محدوده معتبر وارد کن رفیق! 📏')),
      );
      return;
    }

    setState(() => _saving = true);
    await widget.appState.saveProfile(
      newNickname: _nickCtrl.text.trim(),
      newBDay: _bDay,
      newBMonth: _bMonth,
      newBYear: _bYear,
      newGender: _gender,
      newCity: _cityCtrl.text.trim(),
      newEconomicStatus: _econ,
      newHeight: h,
      newWeight: w,
      newGoal: _goal,
      newTimeline: _timeline,
      newWorkoutDays: _workoutDays,
      newAiMemory: _memoryCtrl.text.trim(),
    );
    if (mounted) {
      setState(() => _saving = false);
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('✅ پرونده بدنی ذخیره شد! از این لحظه در تمام پاسخ‌های AI اعمال می‌شه 🚀'),
        ),
      );
    }

/// تب ۸: پنل مدیرعامل (مخصوص متین - همراه با آمار وضعیت اقتصادی کاربران)
class CEOAdminTab extends StatefulWidget {
  final AppState appState;
  const CEOAdminTab({super.key, required this.appState});

  @override
  State<CEOAdminTab> createState() => _CEOAdminTabState();
}

class _CEOAdminTabState extends State<CEOAdminTab> {
  bool _loading = false;
  Map<String, dynamic> _stats = {
    'totalUsers': 0,
    'proUsers': 0,
    'todayAiRequests': 0,
    'econWeak': 0,
    'econMedium': 0,
    'econHigh': 0,
  };

  @override
  void initState() {
    super.initState();
    _fetchAdminStats();
  }

  Future<void> _fetchAdminStats() async {
    setState(() => _loading = true);
    try {
      final res = await http.get(
        Uri.parse('$kServerUrl/api/admin/stats'),
        headers: {
          if (widget.appState.token.isNotEmpty)
            'Authorization': 'Bearer ${widget.appState.token}',
        },
      );
      if (res.statusCode == 200) {
        final data = jsonDecode(utf8.decode(res.bodyBytes));
        if (mounted) {
          setState(() {
            _stats = {
              'totalUsers': data['totalUsers'] ?? 1,
              'proUsers': data['proUsers'] ?? 0,
              'todayAiRequests': data['todayAiRequests'] ?? 0,
              'econWeak': data['econWeak'] ?? 0,
              'econMedium': data['econMedium'] ?? 1,
              'econHigh': data['econHigh'] ?? 0,
            };
          });
        }
      }
    } catch (_) {
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    final isDark = widget.appState.isDark;
    return ListView(
      padding: const EdgeInsets.all(16),
      children: [
        Container(
          padding: const EdgeInsets.all(16),
          decoration: BoxDecoration(
            color: BMColors.card(isDark),
            borderRadius: BorderRadius.circular(22),
            border: Border.all(color: BMColors.neonGreen),
          ),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Row(
                mainAxisAlignment: MainAxisAlignment.spaceBetween,
                children: [
                  Text(
                    '👑 پنل فرماندهی مدیرعامل (متین)',
                    style: TextStyle(
                      fontSize: 16,
                      fontWeight: FontWeight.w900,
                      color: BMColors.text(isDark),
                    ),
                  ),
                  IconButton(
                    onPressed: _fetchAdminStats,
                    icon: const Icon(Icons.refresh, color: BMColors.neonGreen),
                  ),
                ],
              ),
              if (_loading)
                const Padding(
                  padding: EdgeInsets.all(16),
                  child: BodyModBLoader(label: 'در حال دریافت آمار زنده از کلودفلر...'),
                )
              else ...[
                const SizedBox(height: 10),
                _statRow('👥 کل کاربران ثبت‌نامی:', '${_stats['totalUsers']} نفر', isDark),
                _statRow('💎 کاربران دارای اشتراک Pro:', '${_stats['proUsers']} نفر', isDark),
                _statRow('⚡ درخواست‌های AI امروز:', '${_stats['todayAiRequests']} درخواست', isDark),
                const Divider(height: 24),
                Text(
                  '📊 تفکیک وضعیت اقتصادی کاربران (از فرم پروفایل):',
                  style: TextStyle(
                    fontSize: 13.5,
                    fontWeight: FontWeight.w900,
                    color: BMColors.neonBlue,
                  ),
                ),
                const SizedBox(height: 8),
                _statRow('🔸 وضعیت اقتصادی ضعیف:', '${_stats['econWeak']} نفر', isDark),
                _statRow('🔹 وضعیت اقتصادی متوسط:', '${_stats['econMedium']} نفر', isDark),
                _statRow('🌟 وضعیت اقتصادی عالی:', '${_stats['econHigh']} نفر', isDark),
              ],
            ],
          ),
        ),
      ],
    );
  }

  Widget _statRow(String title, String val, bool isDark) {
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 5),
      child: Row(
        mainAxisAlignment: MainAxisAlignment.spaceBetween,
        children: [
          Text(title, style: TextStyle(fontSize: 13, color: BMColors.muted(isDark))),
          Text(
            val,
            style: TextStyle(
              fontSize: 13.5,
              fontWeight: FontWeight.w900,
              color: BMColors.text(isDark),
            ),
          ),
        ],
      ),
    );
  }
}
