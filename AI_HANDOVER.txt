## ۱. محیط توسعه و قوانین تعامل با من (متین)
- همیشه صمیمی و محاوره‌ای جواب بده و از ایموجی استفاده کن.

- تمام تغییرات کد را به صورت اسکریپت‌های یک‌جا (Python یا EOF در Bash) بده تا مستقیم در ترموکس کپی و اجرا کنم.

## ۲. معماری فرانت‌اند (Flutter)
- نام پکیج: ir.matinmohammadi.bodymod
- کل سورس دارت در یک فایل واحد به نام lib/main.dart یکپارچه شده است (فایل‌های part حذف شده‌اند).
- ورود اجباری در ابتدای برنامه حذف شده و برنامه مستقیم وارد MainShellScreen می‌شود.
- آیکون اختصاصی برنامه در icon.png و assets/icon.png قرار دارد.

## ۳. معماری بک‌اند (Cloudflare Worker)
- فایل سرور: worker.js (متصل به دیتابیس KV با نام BODYMOD_DB_KV).
- فایل‌های worker.js، cf.py، .cf_config و *.keystore در .gitignore هستند و هرگز نباید به گیت‌هاب بروند.
- کلیدهای Gemini و تنظیمات سامانه سفیر بله ([https://safir.bale.ai/api/v3/send_message](https://safir.bale.ai/api/v3/send_message) با bot_id: 1711487382 و هدر api-access-key) مستقیم داخل worker.js قرار دارند.
- برای آپلود تغییرات بک‌اند روی کلودفلر از دستور python3 -c "import cf; cf.push()" استفاده می‌شود.

## ۴. سیستم بیلد خودکار (GitHub Actions)
- دو ورک‌فلوی مجزا و موازی داریم:
  1. .github/workflows/build-cafebazaar.yml (با --dart-define=STORE_FLAVOR=bazaar)
  2. .github/workflows/build-myket.yml (با --dart-define=STORE_FLAVOR=myket)
- امضای APK: هر دو ورک‌فلو موظفند فقط و فقط با سکرت RELEASE_KEYSTORE_BASE64 (با alias: bodymod_master_key) فایل‌ها را امضا کنند و تحت هیچ شرایطی کلید جایگزین نسازند (در صورت خطا باید exit 1 بدهند).
- فایل‌های CNAME و index.html در روت گیت‌هاب برای دامنه و اینماد نگهداری می‌شوند.

// نگاشت هوشمند نام مدل‌های بادی‌مود به اندپوینت‌های رسمی گوگل + فال‌بک ضدخطا
function resolveGeminiEndpoints(requestedModel) {
  const m = String(requestedModel || "").toLowerCase();
  if (m.includes("3.8")) {
    return [
      "gemini-2.5-flash",
      "gemini-2.0-flash",
      "gemini-1.5-pro-latest"
    ];
  }
  return [
    "gemini-2.5-flash-lite",
    "gemini-2.0-flash-lite",
    "gemini-2.0-flash"
  ];
}

async function callGeminiWithRotation(promptText, requestedModel, env) {
  const validKeys = GEMINI_KEYS.filter(k => k && k.startsWith("AIza"));
  if (validKeys.length === 0 && env && env.GEMINI_API_KEY) {
    validKeys.push(env.GEMINI_API_KEY);
  }
  if (validKeys.length === 0) {
    throw new Error("هیچ کلید معتبری برای Gemini در ورکر تنظیم نشده است.");
  }

  const modelsToTry = resolveGeminiEndpoints(requestedModel);
  const startIdx = Math.floor(Math.random() * validKeys.length);
  let lastError = "";

  const systemInstruction = `تو مربی اختصاصی، صمیمی، علمی و دلسوز برنامه «بادی‌مود (BodyMod)» هستی.
همیشه به زبان فارسی صمیمی و محاوره‌ای، با ایموجی‌های مناسب و ساختار مرتب پاسخ بده.
تمامی توصیه‌های غذایی و تمرینی را ۱۰۰٪ بر اساس «پرونده کامل بدنی کاربر» (سن، قد، وزن، BMI، هدف، دوره زمانی، تعداد روز تمرین در هفته، شهر، وضعیت اقتصادی و حافظه پزشکی/تغذیه‌ای) شخصی‌سازی کن.
قانون طلایی: اگر BMI کاربر زیر 18.5 است (کمبود وزن)، حتی اگر هدفش عضله‌سازی است، اولویت اولت افزایش وزن اصولی و رسیدن به وزن سالم باشد.`;

  for (const modelName of modelsToTry) {
    for (let i = 0; i < validKeys.length; i++) {
      const apiKey = validKeys[(startIdx + i) % validKeys.length];
      const url = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${apiKey}`;

      try {
        const res = await fetch(url, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            system_instruction: {
              parts: [{ text: systemInstruction }]
            },
            contents: [
              {
                role: "user",
                parts: [{ text: promptText }]
              }
            ],
            generationConfig: {
              temperature: 0.75,
              maxOutputTokens: 2048
            }
          })
        });

        const data = await res.json();
        if (res.ok && data?.candidates?.[0]?.content?.parts?.[0]?.text) {
          return data.candidates[0].content.parts[0].text;
        }
        lastError = data?.error?.message || `HTTP ${res.status}`;
      } catch (err) {
        lastError = err.message;
      }
    }
  }

  throw new Error(`خطا در پاسخگویی هوش مصنوعی: ${lastError}`);
}

async function getUsersIndex(kv) {
  if (!kv) return [];
  const raw = await kv.get("BM_USERS_INDEX");
  if (!raw) return [];
  try {
    return JSON.parse(raw);
  } catch (_) {
    return [];
  }
}

async function registerUserInIndex(kv, userId) {
  if (!kv || !userId) return;
  const list = await getUsersIndex(kv);
  if (!list.includes(userId)) {
    list.push(userId);
    await kv.put("BM_USERS_INDEX", JSON.stringify(list));
  }
}

        let todayAiRequests = 0;

        if (kv) {
          todayAiRequests = parseInt((await kv.get(`STATS_AI_${todayStr}`)) || "0", 10);
          const usersList = await getUsersIndex(kv);
          totalUsers = usersList.length;

          for (const uId of usersList.slice(0, 200)) {
            const raw = await kv.get(`USER_${uId}`);
            if (!raw) continue;

              const u = JSON.parse(raw);
              if (u.isPro) proUsers++;
              const ec = String(u.economicStatus || "متوسط");
              if (ec.includes("ضعیف")) econWeak++;
              else if (ec.includes("عالی")) econHigh++;
              else econMedium++;
            } catch (_) {}

          totalUsers: Math.max(totalUsers, 1),
          proUsers,
          todayAiRequests,
          econWeak,
          econMedium: Math.max(econMedium, totalUsers === 0 ? 1 : econMedium),
          econHigh
        });

      return jsonResp({ error: err.message || "خطای داخلی سرور کلودفلر" }, 500);
